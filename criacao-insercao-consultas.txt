CREATE DATABASE resortporto;
USE resortporto;


CREATE TABLE cliente (
cpf char(11),
nome varchar(150),
email varchar(50),
datanasc date,
pais varchar(50),
estado char(2),
cep varchar(8),
rua varchar(50),
num int,
PRIMARY KEY (cpf)
);

CREATE TABLE telefones (
cpf char(11),
telefone varchar(15),
PRIMARY KEY (cpf, telefone),
FOREIGN KEY (cpf) REFERENCES cliente(cpf)
);

CREATE TABLE quarto (
hotel varchar(100),
numero int(5),
preco float,
lotacao int(2),
PRIMARY KEY  (hotel, numero)
);

CREATE TABLE hospeda (
dataini date,
cpf char(11),
hotel varchar(100),
numero int(5),
pagamento float,
dataout date,
horaout time,
PRIMARY KEY (dataini, cpf, hotel, numero),
FOREIGN KEY (cpf) REFERENCES cliente(cpf),
FOREIGN KEY (hotel, numero) REFERENCES quarto(hotel, numero)
);
